Install

Add the package in Xcode or Package.swift, then wrap your root view.

// Package.swift
.package(url: "https://github.com/promptdora/HIGDesign.git", from: "1.6.0")

// App.swift
import HIGDesign

HIGThemeableView(theme: HIGSystemTheme()) {
    HIGButton("Continue", role: .primary) { }
}
